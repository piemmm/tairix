# What is typed at BASIC's prompt, for `cargo xtask riscos --reference`
# (tools/xtask/src/commands/riscos/session.rs); the format is the crate's
# `corpus::session`. A session prints nothing that depends on the machine's
# memory, and runs LVAR only at the starting @%, whose heading the
# calibration knows.

session lvar
type "NEW"
type "10 DEF PROCp:ENDPROC"
type "20 DEF FNf=1"
type "RUN"
type "A%=5:B%=0:C%=0:D%=0:E%=0:F%=0:G%=0:H%=0:I%=-2147483648:J%=0:K%=0:L%=0:M%=0"
type "N%=0:O%=0:P%=0:Q%=0:R%=0:S%=0:T%=0:U%=0:V%=0:W%=0:X%=0:Y%=0:Z%=0"
type "B=1/3"
type "Cc=1E20"
type "Dd=-1E-6"
type "Ee=123456789"
type "Ff=2^31"
type "Hh=-0.5"
type "S$=\x22\x22"
type "T$=\x22a\x22\x22b\x22"
type "U$=\x22Qz7Kp2Rw9Tm4Xb6Ly3Nc8Vd5Hj1Gf0Su\x22"
type "V$=\x22k4Pw8Zr2Ym6Bt0Qx9\x22"
type "Ww$=\x22j3Hn7Dc1Lv5Fq9Rg2\x22"
type "DIM W(1,2,3),X%(0),Y$(2),Zq(10,10)"
type "zb=1"
type "za=2"
type "zz$=\x22s\x22"
type "zz%=4"
type "zc%=5"
type "Longvariablename123=1"
type "Longintegername%=7"
type "Longstringname$=\x22v\x22"
type "PROCp:X=FNf"
type "LVAR"
type "NEW"
type "10 DEF PROClib:ENDPROC"
type "20 DEF FNlib=2"
type "SAVE \x22HostFS:$.SessLib\x22"
type "NEW"
type "INSTALL \x22HostFS:$.SessLib\x22"
type "LIBRARY \x22HostFS:$.SessLib\x22"
type "PROClib:Q=FNlib"
type "LVAR"
type "*Delete HostFS:$.SessLib"

session edit
type "NEW"
type "10 SYS \x22OS_Module\x22,6,,,1024 TO ,,C%:B%=C%+256:!B%=0"
type "20 FOR I%=0 TO 2 STEP 2:P%=C%:[OPT I%"
type "30 STMFD R13!,{R0-R4}"
type "40 LDR R3,buf:LDR R4,[R3]:ADD R3,R3,#4:MOV R1,#0"
type "50 .loop CMP R4,#700:BHS out"
type "60 LDRB R2,[R0,R1]:ADD R1,R1,#1:CMP R2,#32:MOVLO R2,#124:STRB R2,[R3,R4]:ADD R4,R4,#1:BHS loop"
type "70 .out SUB R3,R3,#4:STR R4,[R3]"
type "80 LDMFD R13!,{R0-R4}:LDR PC,[R13],#4"
type "90 .buf EQUD B%:]:NEXT"
type "100 SYS \x22OS_Claim\x22,5,C%,0"
type "RUN"
type "EDIT 20"
type "PRINT ERR"
type "A$=\x22\x22:FOR I%=0 TO !B%-1:A$+=CHR$?(B%+4+I%):NEXT"
type "PRINT A$=\x22RMEnsure ARMBasicEditor 0 RMLoad System:Modules.BasicEdit|ARMBE|\x22"
type "SYS \x22OS_Release\x22,5,C%,0:SYS \x22OS_Module\x22,7,,C%"

session immediate_assembly

type "O%=C%:P%=&8000:[OPT 7:MOV R0,#1:.lab ADD R0,R0,#1"
type "MOV R1,#2"

type "WIDTH 20"
type "O%=C%:P%=&8000:[OPT 7:MOV R0,#1:REM a remark long enough to wrap at the width"
type "WIDTH 0"
