# The Draw module's screens: each case run by genuine RISC OS 5's BASIC V,
# and what it read and drew recorded, which the replay holds the module to.
# The frame is the VDU drivers' (`tairix_riscos_vdu::corpus::program`):
# `PROCr` records a reading, `FNm` a mode variable, `FNe` an error's
# number, and the screen is saved at the end. The cases here keep to what
# the module matches to the unit: a thick line's width across a slant, a
# long mitre's tip and a dash along a slant differ in low bits
# (`docs/src/desktop/tairix-bridge.md`), and are held to RISC OS in the
# crate's own tests instead. Each case builds its paths with `PROCp`,
# defined in its own lines.

case swis_answer_drawv_with_r8_their_index
10 PROCmode(28):DIM c% 256,l% 4096,p% 512,b% 512,k% 64
20 FOR s%=0 TO 2 STEP 2:P%=c%:[OPT s%
30 STMFD R13!,{R9,R10}:LDR R9,la:LDR R10,[R9]:CMP R10,#3600:ADDLT R10,R10,#36:STRLT R10,[R9]
40 ADDLT R9,R9,R10:ADDLT R9,R9,#4:STMLTIA R9,{R0-R8}:LDMFD R13!,{R9,R10}:MOV PC,R14
50 .la EQUD l%:]:NEXT
60 !p%=2:p%!4=0:p%!8=0:p%!12=8:p%!16=25600:p%!20=0:p%!24=8:p%!28=25600:p%!32=25600:p%!36=5:p%!40=0:p%!44=1024
70 k%!0=p%:k%!4=0:k%!8=0
80 SYS "OS_Claim",&20,c%,0
90 FOR n%=0 TO 16:IF n%<>13 AND n%<>15 THEN PROCcall(n%)
100 NEXT
110 SYS "OS_Release",&20,c%,0
120 GOTO 9000
130 DEFPROCcall(n%):LOCAL r0%,r1%,r4%,r7%,f%,j%,i%,v%
140 !l%=-36:!b%=0:b%!4=400:r1%=0:r4%=0:r7%=0
150 IF n%=6 OR n%=8 OR n%=10 THEN r1%=b%
160 IF n%=12 THEN r4%=k%
170 IF n%=14 THEN r7%=k%
180 SYS &60700+n%,p%,r1%,0,0,r4%,0,0,r7% TO r0%;f%:PROCr("swi"+STR$n%,FNe(r0%,f%))
190 IF !l%<0 THEN ENDPROC
200 FOR j%=0 TO !l% STEP 36:FOR i%=0 TO 8:v%=l%!(4+j%+4*i%):IF v%=p% THEN v%=-2 ELSE IF v%=b% THEN v%=-3 ELSE IF v%=k% THEN v%=-4
